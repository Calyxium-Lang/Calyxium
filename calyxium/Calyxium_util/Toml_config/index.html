Module Calyxium_util.Toml_config

Project and runtime configuration types and parsing utilities.

type project_config = {
  1. name : string;
    (*

    Project name.

    *)
  2. version : string;
    (*

    Project version, e.g., "1.0.0".

    *)
  3. description : string;
    (*

    Short description of the project.

    *)
  4. authors : string list;
    (*

    List of project authors.

    *)
  5. license : string;
    (*

    License identifier, e.g., "MIT".

    *)
  6. homepage : string;
    (*

    URL of the project homepage.

    *)
  7. repository : string;
    (*

    URL of the project's source repository.

    *)
}

Metadata about a project.

type run_config = {

    (*

Complete project configuration.

val parse_line : string -> [> `Entry of string * string | `Ignore | `Section of string ]

parse_line line parses a single line of a configuration file.

  • returns
    • `Entry (key, value) for a key-value pair
    • `Ignore for comments or empty lines
    • `Section name for section headers
val parse_value : string -> [> `Array of string list | `Bool of bool | `String of string ]

parse_value s parses a string value from a configuration file.

  • returns
    • `Array lst if the value represents a list of strings
    • `Bool b if the value represents a boolean
    • `String s for plain string values
val parse_file : string -> config option

parse_file filename parses a TOML or similar configuration file at filename and returns a config if successful.

val create_project : string -> unit

create_project path initializes a new project at path by generating default configuration files and directories.
