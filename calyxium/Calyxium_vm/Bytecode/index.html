type function_info = {
  1. return_type : Calyxium_parser.Ast.Type.t;
    (*

    The function's return type.

    *)
  2. params : Calyxium_parser.Ast.Stmt.parameter list;
    (*

    List of parameters with types.

    *)
  3. bytecode : Opcode.opcode list;
    (*

    Sequence of opcodes for the function.

    *)
  4. body_code : Opcode.opcode array;
    (*

    Array representation of the function body.

    *)
}
