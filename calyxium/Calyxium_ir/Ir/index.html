Module Calyxium_ir.Ir

The intermediate representation (IR) for Calyxium.

type instr = ..

Represents an IR instruction.

type expr = ..

Represents an IR expression.

type var = {
  1. var_name : string;
  2. typ : Calyxium_parser.Ast.Type.t;
  3. value : expr option;
}

Represents a declared variable in the IR.

  • parameter var_name

    The variable's identifier.

  • parameter typ

    The variable's static type.

  • parameter value

    An optional initializer expression.

type func_decl = {
  1. name : string;
  2. is_rec : bool;
  3. parameters : Calyxium_parser.Ast.Stmt.parameter list;
  4. return_type : Calyxium_parser.Ast.Type.t;
  5. body : instr list;
}

Represents a function declaration in the IR.

  • parameter name

    The function's name.

  • parameter is_rec

    Whether the function is recursive.

  • parameter parameters

    The list of function parameters.

  • parameter return_type

    The function's return type.

  • parameter body

    The list of IR instructions forming the function body.

type instr +=
  1. | IR_Expr of expr
    (*

    A standalone expression used as a statement.

    *)
  2. | IR_Block of instr list
    (*

    A block of instructions executed sequentially.

    *)
  3. | IR_FuncDecl of func_decl
    (*

    A function declaration.

    *)

IR instruction constructors.

type expr +=
  1. | IR_Int of Calyxium_parser.Bigint.t
    (*

    Integer literal.

    *)
  2. | IR_Float of float
    (*

    Floating-point literal.

    *)
  3. | IR_String of string
    (*

    String literal.

    *)
  4. | IR_Byte of char
    (*

    Character literal.

    *)
  5. | IR_Bool of bool
    (*

    Boolean literal.

    *)
  6. | IR_Unit
    (*

    The unit value ().

    *)
  7. | IR_Var of string
    (*

    A reference to a variable by name.

    *)
  8. | IR_Binary of expr * Calyxium_parser.Token.t * expr
    (*

    A binary operator application.

    *)
  9. | IR_Unary of Calyxium_parser.Token.t * expr
    (*

    A unary operator application.

    *)
  10. | IR_Call of expr * expr list
    (*

    A function call with a callee expression and a list of arguments.

    *)
  11. | IR_Array of expr list
    (*

    Array literal.

    *)
  12. | IR_Index of expr * expr
    (*

    Array indexing arr.(idx).

    *)
  13. | IR_Tuple of expr list
    (*

    Tuple literal.

    *)
  14. | IR_If of expr * expr * expr option
    (*

    Conditional expression optionally with an else branch.

    *)
  15. | IR_BlockExpr of instr list
    (*

    A block treated as an expression.

    *)
  16. | IR_Ternary of expr * expr * expr
    (*

    Ternary conditional cond ? a : b.

    *)
  17. | IR_Pipeline of expr * expr
    (*

    Pipeline operator lhs |> rhs.

    *)
  18. | IR_Lambda of {
    1. parameters : Calyxium_parser.Ast.Stmt.parameter list;
    2. body : instr list;
    }
    (*

    Anonymous function/lambda expression.

    *)
  19. | IR_Range of expr option * expr option
    (*

    Numeric or iterable range, possibly open-ended.

    *)
  20. | IR_Slice of expr * expr option * expr option
    (*

    Slice operator expr[start:end].

    *)
  21. | IR_Enum of {
    1. name : string;
    2. members : (string * int) list;
    }
    (*

    Enumeration definition.

    *)
  22. | IR_Dot of expr * string
    (*

    Field or member access.

    *)
  23. | IR_Import of {
    1. module_name : string list;
    }
    (*

    Module import expression.

    *)
  24. | IR_Match of expr * (expr option * instr list) list
    (*

    Pattern matching construct.

    *)
  25. | IR_VarDecl of var
    (*

    Variable declaration.

    *)
  26. | IR_MultiVarDecl of {
    1. names : string list;
    2. value : expr;
    3. typ : Calyxium_parser.Ast.Type.t;
    }
    (*

    Multi-variable declaration with a shared initializer.

    *)

val ast_to_ir_stmt : Calyxium_parser.Ast.Stmt.t -> instr

Convert an AST statement into an IR instruction.

val ast_to_ir : Calyxium_parser.Ast.Stmt.t -> instr list

Convert a full AST statement (typically a top-level block or program) into a list of IR instructions.

  • parameter root

    The AST node representing a program or top-level block.

  • returns

    The list of generated IR instructions.
